// Q16. Fix the shopping cart calculation

// function calculateCartTotal(...prices) {
//     let total = 0;
//
//     for (let i = 0; i <= prices.length; i++) {
//         total += prices[i];
//     }
//
//     return total;
// }
//
// console.log(calculateCartTotal(100, 200, 300));

Answer:
The loop uses <= instead of <.
This causes it to access an array position that does not exist, producing undefined and then NaN.

Corrected code:

function calculateCartTotal(...prices) {
    let total = 0;

    for (let i = 0; i < prices.length; i++) {
        total += prices[i];
    }

    return total;
}

console.log(calculateCartTotal(100, 200, 300));

// Output: 600

// Q19. Scope and hoisting

// console.log(a);
// var a = 10;
//
// console.log(b);
// let b = 20;

Answer:

undefined
ReferenceError

Explanation:
var is hoisted and initialized to undefined.
let is hoisted but cannot be accessed before initialization.
This period is called the Temporal Dead Zone (TDZ).

The ReferenceError stops normal execution before let b = 20 is reached.



// SECTION E - JAVASCRIPT INTERVIEW QUESTIONS


// Q20. Difference between == and ===?

Answer:
== allows type conversion before comparison.
=== compares both the value and the type.

Example:

console.log(5 == "5");   // true
console.log(5 === "5");  // false


// Q21. Difference between undefined and null?

Answer:
undefined means a value has not been assigned.
null represents an intentional absence of a value.

Example:

let a;
let b = null;

console.log(a);  // undefined
console.log(b);  // null


// Q22. Difference between for...in and for...of?

Answer:
for...in iterates over enumerable property keys.
for...of iterates over iterable values.

Example:

const arr = ["A", "B"];

for (const key in arr) {
    console.log(key);
}
// 0, 1

for (const value of arr) {
    console.log(value);
}
// A, B


// Q23. Can a function return another function?

Answer:
Yes. This is useful for closures and creating reusable functions.

Example:

function multiplier(x) {
    return function(y) {
        return x * y;
    };
}

const double = multiplier(2);

console.log(double(5));  // 10

Use case:
Creating functions with preset values or settings.


// Q24. What happens if a function has no return statement?

Answer:
It automatically returns undefined.

Example:

function greet() {
    console.log("Hello");
}

console.log(greet());

// Output:
// Hello
// undefined


// Q25. Difference between passing primitive values and objects?

Answer:
JavaScript passes arguments by value.
For objects, the copied value is a reference to the same object, so modifying its properties can affect the original object.

Example:

function update(obj) {
    obj.name = "Rahul";
}

const user = { name: "Aman" };

update(user);

console.log(user.name);  // Rahul


// Q26. Difference between shallow copy and deep copy?

Answer:
A shallow copy copies top-level properties but shares nested objects.
A deep copy also copies nested structures.

Example:

const original = {
    name: "Aman",
    address: { city: "Mathura" }
};

const shallow = { ...original };

shallow.address.city = "Delhi";

console.log(original.address.city);  // Delhi

Explanation:
Both objects share the same nested address object.
structuredClone() can make a deep copy of supported data.


// Q27. Why is NaN !== NaN true?

Answer:
NaN is not equal to any value, including itself.

Example:

console.log(NaN === NaN);          // false
console.log(NaN !== NaN);          // true
console.log(Number.isNaN(NaN));    // true


// Q28. Difference between function declarations and function expressions?

Answer:
Function declarations are hoisted with their definitions.
Function expressions depend on how the variable holding the function is declared and initialized.

Example:

console.log(add(2, 3));  // 5

function add(a, b) {
    return a + b;
}

A const function expression cannot be called before initialization:

const multiply = (a, b) => a * b;

console.log(multiply(2, 3));  // 6


// Q29. When should you use an arrow function instead of a regular function?

Answer:
Use arrow functions for short callbacks and when you want to inherit this from the surrounding context.

Use regular functions for object methods that need their own dynamic this.

Example:

const numbers = [1, 2, 3];

const doubled = numbers.map(n => n * 2);

console.log(doubled);

// Output: [2, 4, 6]
